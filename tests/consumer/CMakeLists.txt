cmake_minimum_required(VERSION 3.25)
project(teder_consumer LANGUAGES C)
find_package(teder 0.1.0 EXACT CONFIG REQUIRED)
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE teder::teder)
enable_testing()
add_test(NAME consumer COMMAND consumer "${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/track-a.flac")
set_tests_properties(consumer PROPERTIES TIMEOUT 20)

if(TARGET teder::notcurses)
	find_path(consumer_notcurses_include notcurses/notcurses.h PATHS "${TEDER_NOTCURSES_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
	find_library(consumer_notcurses_library NAMES notcurses-core PATHS "${TEDER_NOTCURSES_PREFIX}/lib" "${TEDER_NOTCURSES_PREFIX}/lib64" NO_DEFAULT_PATH REQUIRED)
	add_executable(renderer_consumer renderer.c)
	target_include_directories(renderer_consumer PRIVATE "${consumer_notcurses_include}")
	target_link_libraries(renderer_consumer PRIVATE teder::notcurses "${consumer_notcurses_library}")
	add_test(NAME renderer_consumer COMMAND renderer_consumer)
	set_tests_properties(renderer_consumer PROPERTIES TIMEOUT 30)
endif()
