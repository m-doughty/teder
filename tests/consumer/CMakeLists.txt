cmake_minimum_required(VERSION 3.25)
project(teder_consumer LANGUAGES C)
find_package(teder 0.1.0 EXACT CONFIG REQUIRED)
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE teder::teder)
enable_testing()
add_test(NAME consumer COMMAND consumer "${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/track-a.flac")
set_tests_properties(consumer PROPERTIES TIMEOUT 20)
add_executable(example_consumer ../../examples/exercise.c ../../examples/demo.c)
target_compile_features(example_consumer PRIVATE c_std_17)
target_link_libraries(example_consumer PRIVATE teder::teder)
add_test(NAME example_consumer COMMAND example_consumer
	"${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/track-a.flac"
	"${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/track-b.flac")
set_tests_properties(example_consumer PROPERTIES TIMEOUT 45)

if(TARGET teder::notcurses)
	find_path(consumer_notcurses_include notcurses/notcurses.h PATHS "${TEDER_NOTCURSES_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
	find_library(consumer_notcurses_library NAMES notcurses-core PATHS "${TEDER_NOTCURSES_PREFIX}/lib" "${TEDER_NOTCURSES_PREFIX}/lib64" NO_DEFAULT_PATH REQUIRED)
	if(NOT TARGET teder::notcurses_runtime)
		add_library(teder::notcurses_runtime UNKNOWN IMPORTED)
		set_property(TARGET teder::notcurses_runtime PROPERTY IMPORTED_LOCATION "${consumer_notcurses_library}")
	endif()
	add_executable(renderer_consumer renderer.c)
	target_include_directories(renderer_consumer PRIVATE "${consumer_notcurses_include}")
	target_link_libraries(renderer_consumer PRIVATE teder::notcurses "${consumer_notcurses_library}")
	add_test(NAME renderer_consumer COMMAND renderer_consumer)
	set_tests_properties(renderer_consumer PROPERTIES TIMEOUT 30)
	add_executable(visualizer_consumer ../../examples/visualizer.c ../../examples/demo.c)
	target_compile_features(visualizer_consumer PRIVATE c_std_17)
	target_include_directories(visualizer_consumer PRIVATE "${consumer_notcurses_include}")
	target_link_libraries(visualizer_consumer PRIVATE teder::notcurses "${consumer_notcurses_library}")
	if(MINGW)
		target_link_options(visualizer_consumer PRIVATE -municode)
	endif()
	add_test(NAME visualizer_consumer COMMAND visualizer_consumer --check
		"${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/track-a.flac"
		"${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/track-b.flac")
	set_tests_properties(visualizer_consumer PROPERTIES TIMEOUT 45)
endif()

get_target_property(teder_abi_headers teder::teder INTERFACE_INCLUDE_DIRECTORIES)
add_subdirectory(../abi abi)
