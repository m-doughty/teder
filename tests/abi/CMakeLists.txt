find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(variant consumer short-enums)
	if(variant STREQUAL "short-enums" AND MSVC)
		continue() # MSVC has no equivalent compiler switch.
	endif()
	add_executable(abi_${variant} consumer.c)
	target_compile_features(abi_${variant} PRIVATE c_std_17)
	target_link_libraries(abi_${variant} PRIVATE teder::teder)
	if(TARGET teder::notcurses)
		target_link_libraries(abi_${variant} PRIVATE teder::notcurses teder::notcurses_runtime)
		target_compile_definitions(abi_${variant} PRIVATE ABI_WITH_RENDERER)
	endif()
	if(MSVC)
		target_compile_options(abi_${variant} PRIVATE /W4 /WX)
	else()
		target_compile_options(abi_${variant} PRIVATE -Wall -Wextra -Wpedantic -Werror)
		if(variant STREQUAL "short-enums")
			target_compile_options(abi_${variant} PRIVATE -fshort-enums)
		endif()
	endif()
	add_test(NAME abi.${variant} COMMAND abi_${variant})
	set_tests_properties(abi.${variant} PROPERTIES TIMEOUT 30 LABELS abi)
	if(UNIX AND TARGET teder::notcurses_runtime)
		# RUNPATH is not transitive; the host supplies this separate native runtime.
		set(runtime_search LD_LIBRARY_PATH)
		if(APPLE)
			set(runtime_search DYLD_LIBRARY_PATH)
		endif()
		set_tests_properties(abi.${variant} PROPERTIES ENVIRONMENT_MODIFICATION
			"${runtime_search}=path_list_prepend:$<TARGET_FILE_DIR:teder::notcurses_runtime>")
	endif()
endforeach()
set(audit "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/audit.py"
	--headers "${teder_abi_headers}" --nm "${CMAKE_NM}" --objdump "${CMAKE_OBJDUMP}")
add_test(NAME abi.core-exports COMMAND ${audit} --binary "$<TARGET_FILE:teder::teder>")
if(TARGET teder::notcurses)
	add_test(NAME abi.renderer-exports COMMAND ${audit} --renderer --binary "$<TARGET_FILE:teder::notcurses>")
	set_tests_properties(abi.renderer-exports PROPERTIES TIMEOUT 30 LABELS abi)
endif()
add_library(abi_export_leak SHARED leak.c)
target_link_libraries(abi_export_leak PRIVATE teder::teder)
target_compile_definitions(abi_export_leak PRIVATE TEDER_BUILD)
# A static dependency propagates TEDER_STATIC; the fixture itself must still export.
if(WIN32)
	set_target_properties(abi_export_leak PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
add_test(NAME abi.export-control COMMAND ${audit} --negative --binary "$<TARGET_FILE:abi_export_leak>")
set_tests_properties(abi.core-exports abi.export-control PROPERTIES TIMEOUT 30 LABELS abi)
